import { Card, CardGrid } from '@astrojs/starlight/components'

```bash
composer require silarhi/caf-parser
```

<CardGrid>
    <Card title="LA44ZZ payment slips" icon="document">
        Every payment line of the table: beneficiary, period, gross, deduction and net amounts.
    </Card>
    <Card title="Slip metadata" icon="list-format">
        Processing and payment dates, CAF name and address, recipient name and address, reference, BIC, IBAN and total.
    </Card>
    <Card title="Typed results" icon="approve-check">
        Dates as `DateTimeInterface`, amounts as `float`, with French and English formats such as `1 298,50` or
        `1,298.50`.
    </Card>
    <Card title="Strict validation" icon="warning">
        Malformed rows and amounts, and invalid or out-of-range dates, throw a `ParseException` instead of a wrong
        value.
    </Card>
    <Card title="No dependencies" icon="rocket">
        Plain PHP 8.2+, with Windows (`\r\n`) and Unix line endings.
    </Card>
</CardGrid>
